cmake_minimum_required(VERSION 3.20)
project(AssetStreamer VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(STREAMER_BUILD_DEMO "Build the SDL2 streaming demo" ON)
option(STREAMER_BUILD_TESTS "Build GoogleTest tests" ON)
set(STREAMER_SANITIZER "" CACHE STRING "Sanitizer to enable: address, thread, or empty")

find_package(Threads REQUIRED)

add_library(streamer src/AssetSource.cpp src/AssetManager.cpp src/Png.cpp src/TileWorld.cpp)
target_include_directories(streamer PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(streamer PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(streamer PRIVATE /W4)
else()
  target_compile_options(streamer PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(STREAMER_SANITIZER AND NOT MSVC)
  if(STREAMER_SANITIZER STREQUAL "thread")
    set(_san -fsanitize=thread)
  else()
    set(_san -fsanitize=address,undefined)
  endif()
  add_compile_options(${_san} -fno-omit-frame-pointer -g)
  add_link_options(${_san})
endif()

add_executable(frame_benchmark src/frame_benchmark.cpp)
target_link_libraries(frame_benchmark PRIVATE streamer)

add_executable(stream_benchmark src/benchmark.cpp)
target_link_libraries(stream_benchmark PRIVATE streamer)

if(STREAMER_BUILD_DEMO)
  find_package(SDL2 CONFIG QUIET)
  if(NOT SDL2_FOUND)
    find_package(SDL2 REQUIRED)
  endif()
  add_executable(asset_demo src/demo.cpp)
  target_link_libraries(asset_demo PRIVATE streamer)
  if(TARGET SDL2::SDL2)
    target_link_libraries(asset_demo PRIVATE SDL2::SDL2)
    if(TARGET SDL2::SDL2main)
      target_link_libraries(asset_demo PRIVATE SDL2::SDL2main)
    endif()
  else()
    target_include_directories(asset_demo PRIVATE ${SDL2_INCLUDE_DIRS})
    target_link_libraries(asset_demo PRIVATE ${SDL2_LIBRARIES})
  endif()
endif()

if(STREAMER_BUILD_TESTS)
  include(CTest)
  include(FetchContent)
  FetchContent_Declare(googletest
    URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.zip
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(googletest)
  add_executable(streamer_tests tests/ThreadPoolTests.cpp tests/LruCacheTests.cpp tests/AssetManagerTests.cpp tests/PngTests.cpp tests/TileStreamingTests.cpp)
  target_link_libraries(streamer_tests PRIVATE streamer GTest::gtest_main)
  target_compile_definitions(streamer_tests PRIVATE STREAMER_TEST_DATA="${CMAKE_CURRENT_SOURCE_DIR}/tests/data")
  include(GoogleTest)
  gtest_discover_tests(streamer_tests)
endif()
